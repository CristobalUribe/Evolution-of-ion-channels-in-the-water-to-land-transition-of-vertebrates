# First step: Obtain the longest isoform for each gene

# Download proteomes (.faa) from the NCBI FTP, then:

grep ">" Mus_musculus.faa > Mus_musculus.faa_2

awk 'BEGIN{OFS=FS=" "}{print $1}' Mus_musculus.faa_2 > Mus_musculus.faa_ids

# In vi, replace:
:%s/>//g

# After this, we use a script that relies on NCBI datasets (you need to have the datasets executable installed) to assign gene symbols to each NCBI ID in our ID files (e.g., Mus_musculus.faa_ids):

perl script_2.pl > Mus_musculus.faa_ids_2

# To remove empty lines, use:

grep "|" Mus_musculus.faa_ids_2 > Mus_musculus.faa_ids_3

# Change the column order with:

awk 'BEGIN{OFS=FS=" "}{print $3,$2,$1}' Mus_musculus.faa_ids_3 > Mus_musculus.faa_ids_4

# Now you have the IDs for each gene, which will be used to replace headers in the original FASTA using a Python script. This will generate a new file (You need to change file names):

python replace_headers.py

# Using the resulting FASTA file, run the following to obtain the longest isoform:

cat Mus_musculus.fa | awk '/^>/ {if(N>0) printf("\n"); printf("%s\t",$0);N++;next;} {printf("%s",$0);} END {if(N>0) printf("\n");}' | tr "|" "\t" | awk -F '\t' '{printf("%s\t%d\n",$0,length($3));}' | sort -t $'\t' -k1,1 -k4,4nr | sort -t $'\t' -k1,1 -u -s | sed 's/\t/|/' | cut -f 1,2 | tr "\t" "\n" > Mus_musculus_final.fa

# Having obtained the longest protein for each gene, use TMHMM to identify transmembrane proteins:

tmhmm Mus_musculus_final.fa > Mouse_Tmhmm

# Filter transmembrane segments:

awk '/Number of predicted TMHs/ {print $0}' Mouse_Tmhmm > Mouse_Tmhmm_filt

awk 'BEGIN{OFS=FS=" "}{if($7>=2) print $0}' Mouse_Tmhmm_filt > Mouse_Tmhmm_filt2

# Retrieve IDs and sequences:

awk 'BEGIN{OFS=FS=" "}{print $2}' Mouse_Tmhmm_filt2 > Mouse_ids_2

grep -Fwf Mouse_ids_2 Mus_musculus.faa > Full_Mouse_Ids2

python faSomeRecords.py --fasta Mus_musculus.faa --list Full_Mouse_Ids2 -o Mouse_proteins_final.fasta

# Next, run Batch Web CD-Search Tool (CDD v3.21) on the proteins.
# Due to the CD-Search limitation (1000 sequences per request), the FASTA file must be split into smaller parts.

# To filter the output and identify ion channels:

awk 'BEGIN{OFS=FS="\t"}{if($2=="specific" || $2=="superfamily") print $0}' Mouse_hitdata.txt > Mouse_hitdata.txt_filt

awk -F'\t' '{print $0 "\n" $3}' Mouse_hitdata.txt_filt > Mouse_hitdata.txt_filt2

perl script_filt_2.pl > Mouse_hitdata.txt_filt3

python script_filt.py > Mouse_hitdata.txt_filt4

grep "^Q#" Mouse_hitdata.txt_filt4 | awk 'BEGIN {OFS=FS=" "}{print $5}' | sort | uniq -c | awk '{print $2}' > Mouse_hitdata.txt_channels

# or alternatively:

grep "^Q#" Mouse_hitdata.txt_filt4 | awk 'BEGIN {OFS=FS=" "}{print $3}' | sort | uniq -c | awk '{print $2}' > Mouse_hitdata.txt_channels

# At this stage, you will have a file containing IDs or gene symbols, which will be used to retrieve the corresponding FASTA sequences:

grep -Fwf Mouse_hitdata.txt_channels Mouse_proteins_final.fasta > Full_Mouse_Ids

python faSomeRecords.py --fasta Mouse_proteins_final.fasta --list Full_Mouse_Ids -o Mouse_channels.fa

# GUIDANCE3 AND PAML BRANCH-SITE PIPELINE

# ---------------------------------------------------------------------------
# 1. INPUT FILES
# ---------------------------------------------------------------------------

# For each OG, prepare:
#
# 1. One unaligned protein FASTA file.
# 2. One CDS FASTA file corresponding to the same proteins.
# 3. One species tree in Newick format for PAML.
#
# Protein, CDS, and tree labels must identify the same species. CDS sequences
# must be in frame.
#
# A simple project organization is:
#
# project/
# |-- protein_ogs/          Protein FASTA files
# |-- cds_ogs/              CDS files, one directory per OG
# |-- paml_templates/       Species trees, one directory per OG
# |-- scripts/              Scripts used below
# |-- guidance3_analysis/   GUIDANCE3 and filtered codon alignments
# `-- paml_guidance3/       PAML input and results
#
# Example:
#
# protein_ogs/OG1.fasta
# cds_ogs/OG1/cds_species.fasta
# paml_templates/OG1/arbol.tree


# Open a terminal in the project directory and define the folders once.

# 2. DEFINE DIRECTORIES, PROGRAMS AND INPUT LIST

PROJECT_DIR="$(pwd)"
PROTEIN_DIR="$PROJECT_DIR/protein_ogs"
CDS_DIR="$PROJECT_DIR/cds_ogs"
TEMPLATE_DIR="$PROJECT_DIR/paml_templates"
SCRIPT_DIR="$PROJECT_DIR/scripts"
GUIDANCE_DIR="$PROJECT_DIR/guidance3_analysis"
CODON_DIR="$GUIDANCE_DIR/codon_results"
PAML_DIR="$PROJECT_DIR/paml_guidance3"
INPUT_LIST="$PROJECT_DIR/og_protein_files.txt"

G3_BIN="$(command -v guidance3)"
MAFFT_BIN="$(command -v mafft)"
PERL_BIN="$(command -v perl)"
PAL2NAL_BIN="$(command -v pal2nal.pl)"
CODEML_BIN="$(command -v codeml)"

# Required executables must be installed and found before continuing.
# Protein filenames such as OG1.fa, OG1.fasta and OG22.fa_3 are supported.
# For the original 167 OG run, OG22.fa_3 was the corrected input file.
find "$PROTEIN_DIR" -maxdepth 1 -type f -name 'OG*.fa*' -print \
    | sort -V > "$INPUT_LIST"
wc -l "$INPUT_LIST"
head "$INPUT_LIST"
mkdir -p "$GUIDANCE_DIR/results" "$CODON_DIR" "$PAML_DIR"

# 3. RUN GUIDANCE3 (MAFFT L-INS-i, NOT DEFAULT MAFFT)
# The historical run used 100 bootstrap guide-tree replicates, 0.93 cutoff,
# four processes per OG, and up to 20 independent OGs in parallel.
# In the example below, two simultaneous jobs is a conservative default.
# Set MAX_JOBS=20 only when computational resources permit (20 x 4 workers).
# The portable launcher preserves all scientific parameters in the original:
# --msaProgram MAFFT_LINSI --program GUIDANCE3 --bootstraps 100
# --disableConvergence --colCutoff 0.93 --proc_num 4 --mafft <mafft>

export GUIDANCE_ROOT="$GUIDANCE_DIR"
export INPUT_LIST G3_BIN
export MAFFT="$MAFFT_BIN"
export MAX_JOBS="${MAX_JOBS:-2}"
export PROC_NUM=4
export EXPECTED_OGS="$(wc -l < "$INPUT_LIST")"

bash "$SCRIPT_DIR/run_guidance3_linsi_portable.sh" \
    > "$GUIDANCE_DIR/guidance3_launcher.log" 2>&1

# Each OG should produce results/OG_NAME_guidance/ENDS_OK, a BestMSA protein
# alignment (*BestMSA.fasta), and its confidence (*BestMSA_res_pair_col.scr).
# The automatic GUIDANCE3-filtered alignment is NOT the codon-filtering input.
find "$GUIDANCE_DIR/results" -name 'ENDS_OK' -type f | wc -l

# 4. CONVERT BestMSA TO CODONS AND FILTER ALIGNMENT COLUMNS
# prepare_guidance3_codons.py uses PAL2NAL on the FULL BestMSA; it removes an
# entire protein column and corresponding 3-nt codon column if its residue-pair
# confidence score is strictly below 0.93, in ALL species. Missing scores are
# retained and audited as KEEP_UNSCORED, not treated as low confidence.
# Surviving TAA/TAG/TGA codons are masked as NNN, with an audit trail.
# Whole-codon gaps (---) are preserved; partial-gap codons fail validation.
# Input CDS files are under cds_ogs/OG_NAME/cds_species.fasta.

python3 "$SCRIPT_DIR/prepare_guidance3_codons.py" \
    --guidance-root "$GUIDANCE_DIR" \
    --source-codon-root "$CDS_DIR" \
    --output-root "$CODON_DIR" \
    --cutoff 0.93 \
    --perl "$PERL_BIN" \
    --pal2nal "$PAL2NAL_BIN"

# For every OG: OG_BestMSA_filtered_codon.fasta, OG_column_filter.tsv,
# OG_stop_mask.tsv, CODON_FILTER_OK; plus global guidance3_codon_summary.tsv.
find "$CODON_DIR" -name CODON_FILTER_OK -type f | wc -l

# 5. BUILD PAML ALTERNATIVE AND NULL MODELS
# IMPORTANT: template-paml-root must contain:
#   paml_templates/alternative/OG1/codeml.ctl
#   paml_templates/alternative/OG1/arbol.tree
#   paml_templates/null/OG1/arbol.tree
# The two trees must match exactly, and their tips must match the codon alignment.
# Alternative and null use model=2, NSsites=2, cleandata=1.
# Alternative: fix_omega=0, INITIAL omega=0.4 (NOT omega=4).
# Null:        fix_omega=1, omega=1.
# These are starting/control settings, not the fitted omega estimates.

python3 "$SCRIPT_DIR/prepare_guidance3_paml.py" \
    --codon-root "$CODON_DIR" \
    --template-paml-root "$TEMPLATE_DIR" \
    --output-root "$PAML_DIR" \
    --cleandata 1

# The script creates alternative/OG_NAME, null/OG_NAME, PAML_INPUT_OK
# markers, guidance3_paml_input_summary.tsv, and PAML_INPUTS_READY.
# With cleandata=1, codeml subsequently excludes codon columns containing
# gaps or ambiguous characters in any of the analyzed sequences.
find "$PAML_DIR" -name PAML_INPUT_OK -type f | wc -l

# 6. RUN CODEML ON BOTH MODELS
# codeml jobs are independent and usually single-threaded. The historical
# study used --max-jobs 80. Here MAX_CODEML_JOBS defaults to 1 for portability.
# The launcher requires PAML_INPUTS_READY and validates model configurations.
# It is resumable; completed runs receive CODEML_RUN_OK (NOT CODEML_OK).
N_OG="$(wc -l < "$INPUT_LIST")"
MAX_CODEML_JOBS="${MAX_CODEML_JOBS:-1}"

bash "$SCRIPT_DIR/run_guidance3_paml.sh" \
    --paml-root "$PAML_DIR" \
    --codeml "$CODEML_BIN" \
    --expected-ogs "$N_OG" \
    --max-jobs "$MAX_CODEML_JOBS" \
    > "$PAML_DIR/codeml_launcher.log" 2>&1

# There are two expected codeml runs per OG, one alternative and one null.
EXPECTED_MODELS=$((2 * N_OG))
COMPLETED_MODELS="$(find "$PAML_DIR/alternative" "$PAML_DIR/null" \
    -type f -name CODEML_RUN_OK | wc -l)"
printf 'OGs=%s expected_models=%s complete_models=%s\n' \
    "$N_OG" "$EXPECTED_MODELS" "$COMPLETED_MODELS"
# The launcher writes guidance3_codeml_run_summary.tsv (including lnL)
# and CODEML_BATCH_COMPLETE when every expected model is complete.

# 7. EXTRACT STATISTICS AND BEB SITES
# GUIDANCE3/PAML launcher output provides one lnL per model in
# guidance3_codeml_run_summary.tsv. A separate statistical step must calculate
# LRT = max(0, 2 * (lnL_alternative - lnL_null)) per OG, the appropriate
# branch-site test p-values, and a multiple-testing correction (e.g., BH-FDR).
# parse_paml_beb_guidance3.py is specific to the 39 preselected OGs encoded in
# its GENES dictionary (It depends of the PAML results); it only reads alternative-model output and extracts
# site-class MLE tables and starred BEB sites. For that 39-OG analysis only:
 python3 "$SCRIPT_DIR/parse_paml_beb_guidance3.py" "$PAML_DIR/alternative" \
     --output "$PAML_DIR/selected_39_BEB.csv"
# BEB positions are relative to the FINAL FILTERED codon alignment. To map
# them back to the unfiltered alignment, use OG_column_filter.tsv.
